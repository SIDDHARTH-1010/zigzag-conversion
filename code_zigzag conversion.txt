class Solution(object):
    def convert(self, s, numRows):
        # Edge case
        if numRows == 1 or numRows >= len(s):
            return s
        
        # Create rows
        rows = [""] * numRows
        
        current_row = 0
        going_down = False
        
        for char in s:
            rows[current_row] += char
            
            # Change direction
            if current_row == 0 or current_row == numRows - 1:
                going_down = not going_down
            
            # Move up or down
            if going_down:
                current_row += 1
            else:
                current_row -= 1
        
        # Combine all rows
        return "".join(rows)
